import { Card, CardGrid } from '@astrojs/starlight/components'

<CardGrid>
  <Card title="Start here">
    [Install it](/install/), [open a folder as a vault](/vault/), and [find your way around the
    window](/window/). Nothing is imported and nothing is converted: a folder of markdown files is a
    vault from the first minute.
  </Card>
  <Card title="Write and link">
    [Writing](/editor/) saves automatically as you type. [Links](/links/) have a direction: a note
    is a child of another note, or jumps to it. The [plex](/plex/) draws that structure.
  </Card>
  <Card title="Find and ask">
    [Finding](/search/) searches names, words and meaning at once. [The agent](/agent/) replies
    beside the note you are looking at, and can write in the vault when you ask it to.
  </Card>
  <Card title="Recordings and scans">
    A lecture is [transcribed](/recordings/) and a scanned book is [read](/documents/#scanned-books)
    on your own machine. Both become searchable, and a result opens the moment or the page it is on.
  </Card>
  <Card title="Cards">
    [Decks and card models](/cards/) are markdown in the same folder as your notes.
    [Review](/review/) runs in its own window: it shows what is due today, one card at a time.
  </Card>
  <Card title="Make it yours">
    [Settings](/settings/), [themes](/themes/) and the [keyboard](/keyboard/). When the application
    says it cannot do something, [what it means](/trouble/).
  </Card>
</CardGrid>
